ingredients:
  - { qty: 1, unit: "cup", item: "butter, melted" }
  - { qty: 1, unit: "cup", item: "brown sugar" }
  - { qty: 0.5, unit: "cup", item: "white sugar" }
  - { qty: 2, unit: "", item: "eggs" }
  - { qty: 2, unit: "tsp", item: "vanilla" }
  - { qty: 2.25, unit: "cup", item: "all-purpose flour" }
  - { qty: 1, unit: "tsp", item: "baking soda" }
  - { qty: 1, unit: "tsp", item: "salt" }
  - { qty: 2, unit: "cup", item: "chocolate chips" }
steps:
  - "Heat oven to 350°F and line two baking sheets."
  - "Whisk melted butter with both sugars until smooth, then whisk in eggs and vanilla."
  - "Stir in flour, baking soda and salt, then fold in chocolate chips."
  - "Scoop 2-tablespoon balls onto the sheets, 2 inches apart."
  - "Bake 10–11 minutes, until the edges are golden and the centers look slightly underdone. Cool on the sheet 5 minutes."
tip: "Chill the dough 30 minutes for thicker cookies."
