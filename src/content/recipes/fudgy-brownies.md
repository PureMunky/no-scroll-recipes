ingredients:
  - { qty: 0.5, unit: "cup", item: "butter, melted" }
  - { qty: 1, unit: "cup", item: "sugar" }
  - { qty: 2, unit: "", item: "eggs" }
  - { qty: 1, unit: "tsp", item: "vanilla" }
  - { qty: 0.33, unit: "cup", item: "unsweetened cocoa powder" }
  - { qty: 0.5, unit: "cup", item: "all-purpose flour" }
  - { qty: 0.25, unit: "tsp", item: "salt" }
  - { qty: 0.25, unit: "tsp", item: "baking powder" }
  - { qty: 0.5, unit: "cup", item: "chocolate chips (optional)" }
steps:
  - "Heat oven to 350°F and line an 8×8-inch pan with parchment."
  - "Whisk melted butter and sugar, then whisk in eggs and vanilla."
  - "Stir in cocoa, flour, salt and baking powder just until combined. Fold in chips."
  - "Spread in the pan and bake 20–25 minutes, until the center is just set."
  - "Cool completely before cutting into 16 squares."
tip: "Pull them when a toothpick shows a few moist crumbs — that's fudgy, not underdone."
