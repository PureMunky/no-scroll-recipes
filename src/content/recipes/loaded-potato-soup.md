  - { qty: 2.5, unit: "lb", item: "russet potatoes, peeled and cubed" }
  - { qty: 4, unit: "cup", item: "chicken broth" }

  - { qty: 3, unit: "", item: "green onions, sliced" }
steps:
  - "Cook bacon in a large pot over medium until crisp. Remove, leaving 2 tbsp fat."
  - "Cook onion in the fat until soft, 5 minutes, then garlic for 30 seconds."
  - "Add potatoes, broth and salt. Simmer 15 minutes until the potatoes are very tender."
  - "Mash about half the potatoes right in the pot, then stir in milk, sour cream and 1 cup cheese."
  - "Serve topped with bacon, the remaining cheese and green onions."
