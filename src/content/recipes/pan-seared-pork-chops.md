  - { qty: 4, unit: "", item: "boneless pork chops, about 1 inch thick" }
  - { qty: 1, unit: "tsp", item: "salt" }
  - { qty: 0.5, unit: "tsp", item: "black pepper" }
  - { qty: 1, unit: "tsp", item: "paprika" }
  - { qty: 1, unit: "tbsp", item: "neutral oil" }
  - { qty: 2, unit: "tbsp", item: "butter" }
  - { qty: 2, unit: "", item: "garlic cloves, smashed" }
  - { qty: 2, unit: "", item: "sprigs fresh thyme (or ½ tsp dried)" }
steps:
  - "Pat chops dry and season with salt, pepper and paprika."
  - "Heat oil in a skillet over medium-high. Sear chops 4 minutes without moving."
  - "Flip, add butter, garlic and thyme, and baste with the melted butter for 3–4 minutes."
  - "Remove at 145°F and rest 3 minutes before serving."
