ingredients:
  - { qty: 8, unit: "oz", item: "elbow macaroni" }
  - { qty: 2, unit: "tbsp", item: "butter" }
  - { qty: 2, unit: "tbsp", item: "all-purpose flour" }
  - { qty: 2, unit: "cup", item: "milk" }
  - { qty: 0.5, unit: "tsp", item: "dry mustard powder" }
  - { qty: 0.5, unit: "tsp", item: "salt" }
  - { qty: 2, unit: "cup", item: "shredded sharp cheddar (8 oz)" }
steps:
  - "Cook pasta in salted water until just tender. Drain."
  - "In the same pot melt butter over medium, whisk in flour, and cook 1 minute."
  - "Slowly whisk in milk and cook, stirring, until it thickens slightly, 4–5 minutes."
  - "Off heat, stir in mustard, salt and cheese until smooth."
  - "Fold in the pasta. Loosen with a splash of milk if needed."
tip: "Shred the cheese yourself — pre-shredded can make the sauce grainy."
