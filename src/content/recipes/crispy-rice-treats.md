ingredients:
  - { qty: 3, unit: "tbsp", item: "butter" }
  - { qty: 10, unit: "oz", item: "bag mini marshmallows" }
  - { qty: 1, unit: "pinch", item: "salt" }
  - { qty: 0.5, unit: "tsp", item: "vanilla" }
  - { qty: 6, unit: "cup", item: "crisp rice cereal" }
steps:
  - "Grease a 9×13-inch pan."
  - "Melt butter in a large pot over low heat. Add marshmallows and stir until completely melted."
  - "Off heat, stir in salt and vanilla, then the cereal until evenly coated."
  - "Press gently into the pan with a greased spatula. Cool 30 minutes, then cut."
tip: "Press lightly — packing them hard makes them dense."
