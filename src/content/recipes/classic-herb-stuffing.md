  - { qty: 1, unit: "", item: "large onion, diced" }
  - { qty: 3, unit: "", item: "celery ribs, diced" }
  - { qty: 2, unit: "", item: "garlic cloves, minced" }
  - { qty: 1.5, unit: "tsp", item: "dried sage" }
  - { qty: 1, unit: "tsp", item: "dried thyme" }
  - { qty: 1, unit: "tsp", item: "salt" }
  - { qty: 0.5, unit: "tsp", item: "black pepper" }
  - { qty: 0.25, unit: "cup", item: "chopped parsley" }
  - { qty: 2.5, unit: "cup", item: "chicken broth" }
  - { qty: 2, unit: "", item: "eggs" }
steps:
  - "Spread bread cubes on two sheet pans and bake at 300°F for 20 minutes until dry. (Or leave them out overnight.) Raise oven to 375°F and butter a 9×13-inch dish."
  - "Melt butter in a skillet over medium. Cook onion and celery until soft, 8 minutes, then add garlic, sage, thyme, salt and pepper for 1 minute."
  - "In a large bowl toss bread with the vegetables and parsley."
  - "Whisk broth and eggs, pour over the bread, and toss until evenly moistened. Spread in the dish."
  - "Bake covered 25 minutes, then uncovered 20 minutes, until golden on top and 165°F in the center."
tip: "Assemble up to a day ahead and refrigerate; add 10 minutes to the covered bake."
