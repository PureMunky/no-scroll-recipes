ingredients:
  - { qty: 12, unit: "oz", item: "fresh or frozen cranberries" }
  - { qty: 1, unit: "cup", item: "sugar" }
  - { qty: 0.5, unit: "cup", item: "orange juice" }
  - { qty: 0.5, unit: "cup", item: "water" }
  - { qty: 1, unit: "", item: "orange, zested" }
  - { qty: 1, unit: "pinch", item: "salt" }
steps:
  - "Bring sugar, orange juice and water to a simmer in a saucepan, stirring to dissolve."
  - "Add cranberries and simmer 8–10 minutes, until most berries have burst."
  - "Stir in zest and salt."
  - "Cool to room temperature — it thickens a lot as it cools — then refrigerate."
tip: "Keeps 1 week in the fridge, so it's an easy job to cross off early."
