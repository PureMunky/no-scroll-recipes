  - { qty: 1, unit: "cup", item: "all-purpose flour (for crust)" }
  - { qty: 0.25, unit: "cup", item: "powdered sugar, plus more for dusting" }
  - { qty: 0.5, unit: "cup", item: "butter, melted" }
  - { qty: 1, unit: "pinch", item: "salt" }
  - { qty: 2, unit: "", item: "eggs" }
  - { qty: 1, unit: "cup", item: "sugar" }
  - { qty: 2, unit: "tbsp", item: "all-purpose flour (for filling)" }
  - { qty: 0.33, unit: "cup", item: "fresh lemon juice" }
  - { qty: 1, unit: "", item: "lemon, zested" }
steps:
  - "Heat oven to 350°F and line an 8×8-inch pan with parchment."
  - "Mix crust flour, powdered sugar, butter and salt. Press into the pan and bake 15–18 minutes until lightly golden."
  - "Whisk eggs, sugar, filling flour, lemon juice and zest until smooth."
  - "Pour over the hot crust and bake 20–22 minutes, until the center no longer jiggles."
  - "Cool, then chill 1 hour. Dust with powdered sugar and cut into squares."
