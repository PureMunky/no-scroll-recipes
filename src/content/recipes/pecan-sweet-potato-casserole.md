  - { qty: 4, unit: "tbsp", item: "butter, melted (for filling)" }
  - { qty: 2, unit: "", item: "eggs" }
  - { qty: 0.5, unit: "cup", item: "milk" }
  - { qty: 1, unit: "tsp", item: "vanilla" }
  - { qty: 0.5, unit: "tsp", item: "salt" }
  - { qty: 0.5, unit: "tsp", item: "cinnamon" }
  - { qty: 0.75, unit: "cup", item: "brown sugar" }
  - { qty: 0.33, unit: "cup", item: "all-purpose flour" }
  - { qty: 4, unit: "tbsp", item: "butter, melted (for topping)" }
  - { qty: 1, unit: "cup", item: "chopped pecans" }
steps:
  - "Heat oven to 350°F and butter a 9×13-inch dish."
  - "Boil sweet potatoes in salted water until very tender, 15 minutes. Drain well."
  - "Mash with sugar, butter, eggs, milk, vanilla, salt and cinnamon until smooth. Spread in the dish."
  - "Mix brown sugar, flour, butter and pecans and scatter over the top."
  - "Bake 25–30 minutes, until the topping is crisp and the edges bubble."
tip: "Prefer marshmallows? Skip the crumble and cover with mini marshmallows for the last 10 minutes."
