emoji: "🥜"
tags: ["Breakfast", "Make-ahead", "Vegetarian"]
time: 35
servings: 8
ingredients:
  - { qty: 3, unit: "cup", item: "rolled oats" }
  - { qty: 1, unit: "cup", item: "chopped nuts (almonds, pecans or walnuts)" }
  - { qty: 1, unit: "tsp", item: "cinnamon" }
  - { qty: 0.5, unit: "tsp", item: "salt" }
  - { qty: 0.33, unit: "cup", item: "maple syrup" }
  - { qty: 0.33, unit: "cup", item: "olive or coconut oil" }
  - { qty: 1, unit: "tsp", item: "vanilla" }
  - { qty: 0.5, unit: "cup", item: "dried fruit" }
steps:
  - "Heat oven to 325°F and line a sheet pan with parchment."
  - "Mix oats, nuts, cinnamon and salt. Stir in maple syrup, oil and vanilla until everything is coated."
  - "Spread on the pan and press down lightly."
  - "Bake 25 minutes, stirring once halfway, until golden."
  - "Cool completely on the pan — it crisps as it cools — then break up and stir in dried fruit."
tip: "Stored airtight, it keeps 2 weeks."
