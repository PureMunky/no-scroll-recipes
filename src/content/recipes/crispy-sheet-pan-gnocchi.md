servings: 3
ingredients:
  - { qty: 17, unit: "oz", item: "shelf-stable potato gnocchi" }
  - { qty: 1, unit: "", item: "red bell pepper, chopped" }
  - { qty: 1, unit: "", item: "zucchini, chopped" }
  - { qty: 1, unit: "", item: "red onion, cut in wedges" }
  - { qty: 3, unit: "tbsp", item: "olive oil" }
  - { qty: 1, unit: "tsp", item: "Italian seasoning" }
  - { qty: 0.75, unit: "tsp", item: "salt" }
  - { qty: 4, unit: "oz", item: "fresh mozzarella, torn" }
steps:
  - "Heat oven to 450°F."
  - "Toss gnocchi (straight from the pack, uncooked) and vegetables with oil, seasoning and salt on a sheet pan."
  - "Spread in one layer and roast 20 minutes, flipping halfway, until the gnocchi are golden."
  - "Scatter mozzarella over and roast 3 more minutes until melty."
tip: "Don't boil the gnocchi first — roasting them raw is what makes them crisp."
