  - { qty: 1, unit: "", item: "onion, diced" }
  - { qty: 3, unit: "", item: "carrots, sliced" }
  - { qty: 3, unit: "", item: "celery ribs, sliced" }
  - { qty: 2, unit: "", item: "garlic cloves, minced" }
  - { qty: 8, unit: "cup", item: "chicken broth" }
  - { qty: 1, unit: "tsp", item: "dried thyme" }
  - { qty: 1, unit: "", item: "bay leaf" }
  - { qty: 1.5, unit: "lb", item: "boneless chicken breasts" }
  - { qty: 6, unit: "oz", item: "wide egg noodles" }
  - { qty: 1, unit: "tsp", item: "salt" }
  - { qty: 2, unit: "tbsp", item: "chopped parsley" }
steps:
  - "Cook onion, carrots and celery in butter in a large pot over medium until softened, 8 minutes. Add garlic for 30 seconds."
  - "Add broth, thyme, bay leaf and the whole chicken breasts. Simmer gently 15–18 minutes until the chicken reaches 165°F."
  - "Remove chicken, shred it with two forks, and discard the bay leaf."
  - "Add noodles and simmer until tender, 6–8 minutes."
  - "Return chicken, add salt to taste, and stir in parsley."
tip: "If you plan leftovers, cook the noodles separately so they don't soak up all the broth."
