  - { qty: 2, unit: "tbsp", item: "chopped parsley" }

  - { qty: 0.25, unit: "cup", item: "grated parmesan" }
steps:
  - "Heat oven to 400°F."
  - "Mash butter with garlic, parsley and salt."
  - "Split the loaf lengthwise and spread the butter on both cut sides. Sprinkle with parmesan."
  - "Bake cut-side up on a sheet pan 10–12 minutes until golden at the edges."
  - "Slice and serve hot."
