ingredients:
  - { qty: 1, unit: "lb", item: "ziti or penne" }
  - { qty: 1, unit: "lb", item: "Italian sausage, casings removed" }
  - { qty: 24, unit: "oz", item: "jar marinara sauce" }
  - { qty: 15, unit: "oz", item: "ricotta" }
  - { qty: 1, unit: "", item: "egg" }
  - { qty: 2, unit: "cup", item: "shredded mozzarella" }
  - { qty: 0.5, unit: "cup", item: "grated parmesan" }
  - { qty: 0.5, unit: "tsp", item: "salt" }
steps:
  - "Heat oven to 375°F. Cook pasta 2 minutes less than the package says. Drain."
  - "Brown sausage in a skillet, breaking it up. Drain fat and stir in the marinara."
  - "Mix ricotta, egg, half the parmesan and salt."
  - "In a 9×13 dish layer half the pasta, all the ricotta, half the sauce and half the mozzarella. Repeat with the remaining pasta, sauce, mozzarella and parmesan."
  - "Bake covered with foil 20 minutes, then uncovered 15 minutes until bubbly and browned. Rest 10 minutes."
tip: "Assemble up to a day ahead; add 10 minutes to the covered bake if it's cold from the fridge."
