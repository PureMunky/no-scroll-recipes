title: "Apple Crisp"
emoji: "🍎"
tags: ["Dessert", "Baking", "Vegetarian"]
time: 60
servings: 8
ingredients:
  - { qty: 6, unit: "", item: "apples, peeled and sliced (about 2½ lb)" }
  - { qty: 2, unit: "tbsp", item: "sugar" }
  - { qty: 1, unit: "tbsp", item: "lemon juice" }
  - { qty: 1, unit: "tbsp", item: "all-purpose flour (for apples)" }
  - { qty: 1.5, unit: "tsp", item: "cinnamon, divided" }
  - { qty: 0.75, unit: "cup", item: "all-purpose flour (for topping)" }
  - { qty: 0.75, unit: "cup", item: "rolled oats" }
  - { qty: 0.75, unit: "cup", item: "brown sugar" }
  - { qty: 0.25, unit: "tsp", item: "salt" }
  - { qty: 0.5, unit: "cup", item: "cold butter, cubed" }
steps:
  - "Heat oven to 375°F."
  - "Toss apples with sugar, lemon juice, 1 tbsp flour and 1 tsp cinnamon. Spread in a 9-inch square baking dish."
  - "Mix the topping flour, oats, brown sugar, salt and remaining cinnamon. Rub in butter with your fingers until clumpy."
  - "Scatter topping over the apples."
  - "Bake 40–45 minutes, until golden and bubbling at the edges. Cool 10 minutes."
tip: "Serve warm with vanilla ice cream."
