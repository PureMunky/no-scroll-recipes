ingredients:
  - { qty: 1, unit: "lb", item: "chicken breast, cut in 1-inch pieces" }
  - { qty: 1, unit: "tsp", item: "salt" }
  - { qty: 2, unit: "tbsp", item: "olive oil" }
  - { qty: 1, unit: "", item: "shallot or small onion, minced" }
  - { qty: 3, unit: "", item: "garlic cloves, minced" }
  - { qty: 1.5, unit: "cup", item: "orzo" }
  - { qty: 3.5, unit: "cup", item: "chicken broth" }
  - { qty: 1, unit: "", item: "lemon, zested and juiced" }
  - { qty: 2, unit: "cup", item: "baby spinach" }
  - { qty: 0.33, unit: "cup", item: "grated parmesan" }
steps:
  - "Season chicken with half the salt. Brown in oil in a deep skillet over medium-high, 5 minutes. Remove."
  - "Cook shallot and garlic 1 minute, then stir in orzo to toast for 1 minute."
  - "Add broth and the remaining salt, bring to a simmer, and cook 8–10 minutes, stirring often, until the orzo is tender."
  - "Return chicken with lemon zest and juice and cook 2 minutes until the chicken reaches 165°F."
  - "Stir in spinach and parmesan until wilted and creamy."
