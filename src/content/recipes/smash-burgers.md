ingredients:
  - { qty: 1.5, unit: "lb", item: "80/20 ground beef" }
  - { qty: 1, unit: "tsp", item: "salt" }
  - { qty: 0.5, unit: "tsp", item: "black pepper" }
  - { qty: 8, unit: "", item: "slices American cheese" }
  - { qty: 4, unit: "", item: "burger buns, toasted" }
  - { qty: 0, unit: "", item: "pickles, onion, lettuce, sauce, to serve" }
steps:
  - "Divide beef into 8 loose balls (3 oz each). Don't pack them."
  - "Heat a cast-iron skillet or griddle over high until smoking."
  - "Add 2–3 balls, smash flat with a sturdy spatula, and season with salt and pepper."
  - "Cook 2 minutes until deeply browned, scrape up, flip, add cheese, and cook 1 minute more."
  - "Stack two patties per bun with toppings."
tip: "A sheet of parchment between the spatula and the meat keeps it from sticking when you smash."
