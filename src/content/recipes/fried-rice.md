  - { qty: 3, unit: "cup", item: "cold cooked rice" }
  - { qty: 2, unit: "tbsp", item: "neutral oil" }
  - { qty: 3, unit: "", item: "eggs, beaten" }
  - { qty: 1, unit: "cup", item: "frozen peas and carrots" }
  - { qty: 3, unit: "", item: "green onions, sliced" }
  - { qty: 2, unit: "", item: "garlic cloves, minced" }
  - { qty: 3, unit: "tbsp", item: "soy sauce" }
  - { qty: 1, unit: "tsp", item: "sesame oil" }
steps:
  - "Heat 1 tbsp oil in a large skillet over high. Scramble eggs until just set, then move to a plate."
  - "Add the remaining oil, garlic and the white parts of the green onions for 30 seconds."
  - "Add rice and press it flat. Let it sizzle 1–2 minutes without stirring, then toss."
  - "Add peas and carrots, soy sauce and sesame oil. Stir-fry 2 minutes."
  - "Fold the eggs back in and top with the green onion tops."
tip: "Day-old rice is the secret — fresh rice turns mushy."
