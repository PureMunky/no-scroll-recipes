  - { qty: 8, unit: "", item: "slices sourdough bread" }
  - { qty: 2, unit: "tbsp", item: "butter, softened" }
  - { qty: 0.25, unit: "cup", item: "mayonnaise" }
  - { qty: 0.5, unit: "cup", item: "cranberry sauce" }
  - { qty: 12, unit: "oz", item: "sliced leftover turkey" }
  - { qty: 4, unit: "", item: "slices Swiss or provolone cheese" }
  - { qty: 1, unit: "cup", item: "leftover stuffing, warmed (optional)" }
steps:
  - "Butter one side of each bread slice."
  - "On the unbuttered sides, spread mayo on 4 slices and cranberry sauce on the other 4."
  - "Layer turkey, cheese and stuffing on the mayo slices and close the sandwiches, buttered sides out."
  - "Cook in a skillet over medium-low, 3–4 minutes per side, until golden and the cheese has melted."
