emoji: "🥔"

time: 35
servings: 6
ingredients:
  - { qty: 3, unit: "lb", item: "Yukon Gold potatoes, peeled and cubed" }
  - { qty: 6, unit: "", item: "garlic cloves, peeled" }
  - { qty: 1, unit: "tbsp", item: "salt (for the water)" }
  - { qty: 4, unit: "tbsp", item: "butter" }
  - { qty: 0.75, unit: "cup", item: "milk, warmed" }
  - { qty: 0.5, unit: "cup", item: "sour cream" }
  - { qty: 1, unit: "tsp", item: "salt" }
steps:
  - "Put potatoes and garlic in a pot, cover with cold water by an inch, and add 1 tbsp salt."
  - "Boil, then simmer 15–18 minutes until a knife slides in easily."
  - "Drain well and return to the pot over low heat for 1 minute to dry."
  - "Mash with butter, then stir in warm milk, sour cream and salt."
tip: "Starting potatoes in cold water cooks them evenly all the way through."
